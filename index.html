async function submitRequest(){ const level = document.querySelector('.req-lv.sel')?.dataset.l; const types = [...document.querySelectorAll('.req-type.sel')].map(x=>x.textContent).join(', '); const note = document.getElementById('reqNote').value.trim(); if(!level){ showToast('⚠️ Pilih level dulu'); return; } try { const { data: { session } } = await window._supabase.auth.getSession(); if (!session) { showToast('❌ Not logged in'); return; } const usr = window._userDb; const name = usr.display_name || usr.full_name || session.user.email; const email = session.user.email; // Simple: just send with basic chat_id strategy // If user has never chatted bot, use uid as fallback const chat_id = Math.floor(Math.random() * 1000000000); // Placeholder, will fail gracefully const response = await fetch('https://jlpt-auth-bot-production.up.railway.app/request-soal', { method: 'POST', headers: {'Content-Type': 'application/json'}, body: JSON.stringify({ uid: session.user.id, name: name, email: email, level: level, types: types || 'All', note: note, chat_id: chat_id }) }); const data = await response.json(); if (response.ok && data.ok) { showToast('✅ Request terkirim ke admin!'); closeModal('requestModal'); // Reset document.querySelectorAll('.req-lv').forEach(x => x.classList.remove('sel')); document.querySelectorAll('.req-type').forEach(x => x.classList.remove('sel')); document.getElementById('reqNote').value = ''; } else { const errorMsg = data.error || 'Gagal mengirim request'; if (errorMsg.includes('chat_id')) { showToast('⚠️ Chat bot dulu (@jlpt_soal_bot) dengan /start, baru request lagi'); } else { showToast('❌ Error: ' + errorMsg); } } } catch (err) { console.error('Request error:', err); showToast('❌ Network error: ' + err.message); } }
